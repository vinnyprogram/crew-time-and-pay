/** * Crew Time & Pay -- Apps Script backend * * SETUP: * 1. In your Google Sheet: Extensions -> Apps Script. * 2. Delete any starter code, paste this whole file in, and Save. * 3. Click Deploy -> New deployment -> gear icon -> "Web app". * Execute as: Me * Who has access: Anyone * Click Deploy, authorize it, and copy the Web app URL (ends in /exec). * 4. Paste that URL into GAS_URL near the top of index_gsheets.html. * * Your spreadsheet needs exactly 4 tabs with these header rows (row 1): * * Employees: employee_number | name | daily_rate | standard_hours | color * TimeEntries: id | work_date | employee_number | start_time | finish_time | hours_worked | hourly_rate | amount_earned | gasolina | notes (id is generated automatically; finish_time is blank while someone is clocked in) * Payments: payment_date | employee_number | wages_amount | gas_amount | covers_from | covers_to (one row per payment; gas_amount is entered manually, not tied to any day's entries) * Expenses: id | date | category | description | vendor | qty | unit_cost | total | payment_status | payment_date | notes | receipt_url * ExpensePayments: expense_id | payment_date | amount (one row per installment paid toward an expense; expense_id must match an Expenses row's id) (id is generated automatically -- leave it blank when adding rows by hand) * * IMPORTANT: select the work_date/start_time/finish_time/payment_date/date * columns in each sheet and set Format -> Number -> Plain text BEFORE * entering any data. Otherwise Sheets "helpfully" reformats dates/times * and the app's calculations will read them back wrong. */ function doGet(e) { var action = e.parameter.action; if (action === 'all') { // Quietly remove exact duplicate time entries before sending data to the // app. Never allowed to block loading -- if anything goes wrong, skip it. try { var lock = LockService.getScriptLock(); if (lock.tryLock(5000)) { try { cleanDuplicateEntries_(); SpreadsheetApp.flush(); } finally { lock.releaseLock(); } } } catch (err) { /* ignore -- loading data matters more */ } return jsonOut({ employees: sheetToObjects('Employees'), entries: sheetToObjects('TimeEntries'), payments: sheetToObjects('Payments'), expenses: sheetToObjects('Expenses'), expensePayments: sheetToObjects('ExpensePayments') }); } return jsonOut({ error: 'Unknown action: ' + action }); } function doPost(e) { try { var body = JSON.parse(e.postData.contents); var action = body.action; if (action === 'addEntry') { var entry = body.entry; // Lock so two taps arriving at the same moment can't both slip through. var lock = LockService.getScriptLock(); lock.waitLock(20000); try { var dup = findDuplicateEntry_(entry); if (dup) { // Same shift already saved (repeat tap, or a retry after weak signal) -- // don't add another row, just hand back the one that's already there. if (dup.sameShift) return jsonOut({ ok: true, id: dup.row.id, duplicate: true }); return jsonOut({ error: 'J\u00e1 existe um registro nesse hor\u00e1rio para esse dia.' }); } entry.id = Utilities.getUuid(); // lets a later clock-out (or edit) find this exact row appendRow('TimeEntries', entry); SpreadsheetApp.flush(); } finally { lock.releaseLock(); } if (entry.finish_time === '') { notifyClockEvent_('in', entry.employee_number, entry.start_time, entry.work_date); } return jsonOut({ ok: true, id: entry.id }); } else if (action === 'updateEntry') { updateRowByKey('TimeEntries', 'id', body.id, body.patch); if (body.patch && body.patch.finish_time !== undefined) { var entries = sheetToObjects('TimeEntries'); var match = null; for (var j = 0; j < entries.length; j++) { if (entries[j].id === body.id) { match = entries[j]; break; } } if (match) notifyClockEvent_('out', match.employee_number, body.patch.finish_time, match.work_date); } } else if (action === 'addEmployee') { appendRow('Employees', body.employee); } else if (action === 'updateEmployee') { if (body.patch && body.patch.active !== undefined) ensureColumn_('Employees', 'active'); updateRowByKey('Employees', 'employee_number', body.employee_number, body.patch); } else if (action === 'addPayment') { appendRow('Payments', body.payment); } else if (action === 'addExpense') { var expense = body.expense; expense.id = Utilities.getUuid(); // stable id so a receipt can be attached later if (expense.receipt_data) { try { expense.receipt_url = saveReceiptToDrive(expense.receipt_data, expense.receipt_type, expense.receipt_name); } catch (err) { expense.receipt_url = ''; // don't fail the whole expense save just because the upload hiccuped } } appendRow('Expenses', expense); return jsonOut({ ok: true, id: expense.id }); } else if (action === 'attachReceipt') { var receiptUrl = saveReceiptToDrive(body.receipt_data, body.receipt_type, body.receipt_name); updateRowByKey('Expenses', 'id', body.id, { receipt_url: receiptUrl }); } else if (action === 'updateExpense') { updateRowByKey('Expenses', 'id', body.id, body.patch); } else if (action === 'addExpensePayment') { appendRow('ExpensePayments', body.payment); } else if (action === 'deleteEntry') { var delLock = LockService.getScriptLock(); delLock.waitLock(20000); try { deleteRowByKey_('TimeEntries', 'id', body.id, 'deleted in Admin'); } finally { delLock.releaseLock(); } } else { return jsonOut({ error: 'Unknown action: ' + action }); } return jsonOut({ ok: true }); } catch (err) { return jsonOut({ error: err.message }); } } /* ---------------- helpers ---------------- */ // Adds a header column to a sheet if it isn't there yet (used for the // Employees "active" column, so it never has to be typed in by hand). function ensureColumn_(name, header) { var sheet = getSheet_(name); var lastCol = sheet.getLastColumn(); var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0]; if (headers.indexOf(header) === -1) sheet.getRange(1, lastCol + 1).setValue(header); } /** * Auto-cleanup of duplicate time entries (e.g. the same Clock In saved * several times because of weak signal). Only touches rows the app created * (they have an id) and only removes true repeats -- same employee, same day, * same start time: * - extra OPEN copies are removed (if one copy was already clocked out, * every open copy goes; otherwise the first open one is kept) * - CLOSED copies are removed only if finish time and amount also match * Entries that overlap but have different times are NOT touched -- the app * can't know which one is right, so Admin shows a warning for those instead. * Nothing is lost: every removed row is copied to a "RemovedEntries" tab. */ function cleanDuplicateEntries_() { var sheet = getSheet_('TimeEntries'); var values = sheet.getDataRange().getValues(); if (values.length < 3) return 0; var headers = values[0]; var cId = headers.indexOf('id'), cEmp = headers.indexOf('employee_number'), cDate = headers.indexOf('work_date'), cStart = headers.indexOf('start_time'), cFinish = headers.indexOf('finish_time'), cAmt = headers.indexOf('amount_earned'); if ([cId, cEmp, cDate, cStart, cFinish].indexOf(-1) > -1) return 0; var isBlank = function (v) { return v === '' || v === null || v === undefined; }; var groups = {}; for (var r = 1; r < values.length; r++) { var row = values[r]; if (isBlank(row[cId])) continue; // hand-added / imported rows are never touched var start = normalizeCell('start_time', row[cStart]); if (isBlank(start) || isNaN(Number(start))) continue; var key = String(row[cEmp]) + '|' + normalizeCell('work_date', row[cDate]) + '|' + Number(start).toFixed(2); (groups[key] = groups[key] || []).push(r); } var toRemove = []; Object.keys(groups).forEach(function (key) { var rows = groups[key]; if (rows.length < 2) return; var hasClosed = rows.some(function (r) { return !isBlank(values[r][cFinish]); }); var keptOpen = false, keptClosed = []; rows.forEach(function (r) { var fin = values[r][cFinish]; if (isBlank(fin)) { if (hasClosed || keptOpen) toRemove.push(r); else keptOpen = true; } else { var sig = Number(normalizeCell('finish_time', fin)).toFixed(2) + '|' + (cAmt > -1 ? Number(values[r][cAmt] || 0).toFixed(2) : ''); if (keptClosed.indexOf(sig) > -1) toRemove.push(r); else keptClosed.push(sig); } }); }); if (!toRemove.length) return 0; archiveRows_(headers, toRemove.map(function (r) { return values[r]; }), 'auto: duplicate'); toRemove.sort(function (a, b) { return b - a; }) // bottom-up so row numbers don't shift .forEach(function (r) { sheet.deleteRow(r + 1); }); return toRemove.length; } function deleteRowByKey_(name, keyField, keyValue, reason) { var sheet = getSheet_(name); var values = sheet.getDataRange().getValues(); var headers = values[0]; var keyCol = headers.indexOf(keyField); if (keyCol === -1) throw new Error('Column not found: ' + keyField); for (var r = 1; r < values.length; r++) { if (String(values[r][keyCol]) === String(keyValue)) { archiveRows_(headers, [values[r]], reason); sheet.deleteRow(r + 1); return; } } throw new Error('No row found where ' + keyField + ' = ' + keyValue); } // Copies removed rows into a "RemovedEntries" tab (created automatically the // first time) so anything cleaned up can always be checked or restored. function archiveRows_(headers, rows, reason) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('RemovedEntries'); if (!sheet) { sheet = ss.insertSheet('RemovedEntries'); sheet.appendRow(['removed_at', 'reason'].concat(headers)); } var stamp = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm'); rows.forEach(function (row) { sheet.appendRow([stamp, reason].concat(row)); }); } /** * Server-side duplicate guard for new time entries. Returns null if the new * entry is fine to save, otherwise { row, sameShift }: * sameShift = true -> it's a repeat of something already saved (e.g. a * second Clock In while a shift is still open, or the * exact same manual entry again) -- safe to ignore. * sameShift = false -> it overlaps a different existing shift that day. */ function findDuplicateEntry_(entry) { var rows = sheetToObjects('TimeEntries'); var isOpen = function (v) { return v === '' || v === null || v === undefined; }; var newOpen = isOpen(entry.finish_time); var newStart = Number(entry.start_time); var newEnd = newOpen ? 24 : Number(entry.finish_time); for (var i = 0; i < rows.length; i++) { var r = rows[i]; if (String(r.employee_number) !== String(entry.employee_number)) continue; var rOpen = isOpen(r.finish_time); // Clocking in while this person already has an open shift = a repeat tap. if (newOpen && rOpen) return { row: r, sameShift: true }; if (String(r.work_date) !== String(entry.work_date)) continue; var rStart = Number(r.start_time); var rEnd = rOpen ? 24 : Number(r.finish_time); if (newStart < rEnd && rStart < newEnd) { var identical = !newOpen && !rOpen && rStart === newStart && rEnd === newEnd; return { row: r, sameShift: identical }; } } return null; } function getSheet_(name) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name); if (!sheet) throw new Error('Sheet not found: ' + name); return sheet; } function sheetToObjects(name) { var sheet = getSheet_(name); var values = sheet.getDataRange().getValues(); if (values.length < 2) return []; var headers = values[0]; return values.slice(1) .filter(function (row) { return row.some(function (c) { return c !== '' && c !== null; }); }) .map(function (row) { var obj = {}; headers.forEach(function (h, i) { obj[h] = normalizeCell(h, row[i]); }); return obj; }); } /** * Safety net: if a "date" or "time" column ever gets auto-converted by * Sheets into a real Date object (e.g. Plain Text formatting was missed * before typing), rebuild it into the plain string/number the app * expects, instead of letting a raw Date leak out as garbled JSON. */ function normalizeCell(header, value) { if (!(value instanceof Date)) return value; var h = header.toLowerCase(); var isDateField = h.indexOf('date') > -1 || h === 'covers_from' || h === 'covers_to'; if (isDateField) { return Utilities.formatDate(value, 'Etc/GMT', 'yyyy-MM-dd'); } // start_time / finish_time got auto-converted to a time-of-day value -- rebuild as a decimal hour (7, 17.5, ...) var hh = Number(Utilities.formatDate(value, 'Etc/GMT', 'H')); var mm = Number(Utilities.formatDate(value, 'Etc/GMT', 'm')); return Math.round((hh + mm / 60) * 100) / 100; } function appendRow(name, obj) { var sheet = getSheet_(name); var headers = sheet.getDataRange().getValues()[0]; var row = headers.map(function (h) { return (obj[h] !== undefined ? obj[h] : ''); }); sheet.appendRow(row); } function updateRowByKey(name, keyField, keyValue, patch) { var sheet = getSheet_(name); var values = sheet.getDataRange().getValues(); var headers = values[0]; var keyCol = headers.indexOf(keyField); if (keyCol === -1) throw new Error('Column not found: ' + keyField); for (var r = 1; r < values.length; r++) { if (String(values[r][keyCol]) === String(keyValue)) { Object.keys(patch).forEach(function (field) { var col = headers.indexOf(field); if (col > -1) sheet.getRange(r + 1, col + 1).setValue(patch[field]); }); return; } } throw new Error('No row found where ' + keyField + ' = ' + keyValue); } /** * Saves a receipt (photo or PDF) sent from the app into a "Crew Time and * Pay Receipts" folder in your Google Drive, makes it link-viewable, and * returns that link. Only the link is stored in the Expenses sheet -- * the actual file data never touches the spreadsheet. */ function saveReceiptToDrive(dataUrl, mimeType, fileName) { var base64 = dataUrl.indexOf(',') > -1 ? dataUrl.split(',')[1] : dataUrl; var bytes = Utilities.base64Decode(base64); var blob = Utilities.newBlob(bytes, mimeType || 'application/octet-stream', fileName || ('receipt_' + new Date().getTime())); var folder = getOrCreateFolder_('Crew Time and Pay Receipts'); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return file.getUrl(); } function getOrCreateFolder_(name) { var folders = DriveApp.getFoldersByName(name); if (folders.hasNext()) return folders.next(); return DriveApp.createFolder(name); } /** * Emails a clock in/out notification. Set NOTIFY_EMAIL below to a specific * address, or leave it blank to notify the script owner's own account. */ var NOTIFY_EMAIL = 'gvvini.carvalho@gmail.com'; // e.g. 'you@gmail.com' -- blank uses your own Google account automatically function notifyClockEvent_(action, employeeNumber, decimalTime, workDate) { try { var to = NOTIFY_EMAIL || Session.getEffectiveUser().getEmail(); var name = getEmployeeName_(employeeNumber); var timeLabel = formatDecimalTimeForEmail_(decimalTime); var actionLabel = (action === 'in') ? 'Clocked IN' : 'Clocked OUT'; var subject = actionLabel + ': ' + name + ' (' + timeLabel + ')'; var body = name + ' ' + actionLabel + ' at ' + timeLabel + ' on ' + workDate + '.'; MailApp.sendEmail(to, subject, body); } catch (err) { // Never let a failed email break the actual clock in/out save. } } function getEmployeeName_(employeeNumber) { var employees = sheetToObjects('Employees'); for (var i = 0; i < employees.length; i++) { if (String(employees[i].employee_number) === String(employeeNumber)) return employees[i].name; } return 'Employee ' + employeeNumber; } function formatDecimalTimeForEmail_(dec) { var h = Math.floor(Number(dec)); var m = Math.round((Number(dec) - h) * 60); return (h < 10 ? '0' : '') + h + ':' + (m < 10 ? '0' : '') + m; } function jsonOut(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
